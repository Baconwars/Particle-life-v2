// ============================================================
// UPDATE SIMULATION
// ============================================================

void updateSimulation()
{
&#x20;   if (
&#x20;       !simulationRunning ||
&#x20;       !gpuInitialized ||
&#x20;       gpuCellCount <=
&#x20;       0
&#x20;       )
&#x20;   {
&#x20;       return;
&#x20;   }

&#x20;   updateGPUSettings();

&#x20;   if (
&#x20;       trueSpeedEnabled
&#x20;       )
&#x20;   {
&#x20;       simulationStepGPU();
&#x20;   }
&#x20;   else
&#x20;   {
&#x20;       stepAccumulator +=
&#x20;           static\_cast\<double>(
&#x20;               processSpeed
&#x20;               ) /
&#x20;           60.0;

&#x20;       int steps =
&#x20;           static\_cast\<int>(
&#x20;               stepAccumulator
&#x20;               );

&#x20;       stepAccumulator -=
&#x20;           steps;

&#x20;       if (
&#x20;           steps <
&#x20;           1 &&
&#x20;           processSpeed >=
&#x20;           60
&#x20;           )
&#x20;       {
&#x20;           steps =
&#x20;               1;
&#x20;       }

&#x20;       steps =
&#x20;           std::min(
&#x20;               steps,
&#x20;               10
&#x20;           );

&#x20;       for (
&#x20;           int i = 0;
&#x20;           i < steps;
&#x20;           ++i
&#x20;           )
&#x20;       {
&#x20;           simulationStepGPU();
&#x20;       }
&#x20;   }

&#x20;   InvalidateRect(
&#x20;       mainWindow,
&#x20;       nullptr,
&#x20;       FALSE
&#x20;   );
}

// ============================================================
// SEED HASH
// ============================================================

{

&#x20;       2166136261u;

&#x20;   for (
&#x20;       int i = 0;
&#x20;       text[
&#x20;           i
&#x20;       ] != '\0';
&#x20;       ++i
&#x20;               )
&#x20;   {
&#x20;       unsigned char c =
&#x20;           static\_cast\<unsigned char>(
&#x20;               text[
&#x20;                   i
&#x20;               ]
&#x20;               );

&#x20;       if (
&#x20;           c >=
&#x20;           'A' &&
&#x20;           c <=
&#x20;           'Z'
&#x20;           )
&#x20;       {
&#x20;           c =
&#x20;               static\_cast\<unsigned char>(
&#x20;                   c +
&#x20;                   (
&#x20;                       'a' -
&#x20;                       'A'
&#x20;                       )
&#x20;                   );
&#x20;       }

&#x20;       hash ^=
&#x20;           static\_cast\<unsigned int>(
&#x20;               c
&#x20;               );

&#x20;       hash \*=
&#x20;           16777619u;
&#x20;   }

&#x20;   if (
&#x20;       hash ==
&#x20;       0u
&#x20;       )
&#x20;   {
&#x20;       hash =
&#x20;           0xA341316Cu;
&#x20;   }

&#x20;   return hash;
}

// ============================================================
// APPLY SEED
// ============================================================

void applySeedSettings()
{
&#x20;   char seedText[
&#x20;       RANDOM\_SEED\_LENGTH +
&#x20;           1
&#x20;   ] = {};

&#x20;       if (
&#x20;           seedEdit
&#x20;           )
&#x20;       {
&#x20;           GetWindowTextA(
&#x20;               seedEdit,

&#x20;               seedText,

&#x20;               static\_cast\<int>(
&#x20;                   sizeof(
&#x20;                       seedText
&#x20;                       )
&#x20;                   )
&#x20;           );
&#x20;       }

&#x20;       char cleanSeed[
&#x20;           RANDOM\_SEED\_LENGTH +
&#x20;               1
&#x20;       ] = {};

&#x20;           int cleanLength =
&#x20;               0;

&#x20;           for (
&#x20;               int i = 0;

&#x20;               seedText[
&#x20;                   i
&#x20;               ] != '\0' &&
&#x20;               cleanLength <
&#x20;                       RANDOM\_SEED\_LENGTH;

&#x20;                       ++i
&#x20;                       )
&#x20;           {
&#x20;               unsigned char c =
&#x20;                   static\_cast\<unsigned char>(
&#x20;                       seedText[
&#x20;                           i
&#x20;                       ]
&#x20;                       );

&#x20;               bool isNumber =
&#x20;                   c >=
&#x20;                   '0' &&
&#x20;                   c <=
&#x20;                   '9';

&#x20;               bool isLower =
&#x20;                   c >=
&#x20;                   'a' &&
&#x20;                   c <=
&#x20;                   'z';

&#x20;               bool isUpper =
&#x20;                   c >=
&#x20;                   'A' &&
&#x20;                   c <=
&#x20;                   'Z';

&#x20;               if (
&#x20;                   isNumber ||
&#x20;                   isLower ||
&#x20;                   isUpper
&#x20;                   )
&#x20;               {
&#x20;                   cleanSeed[
&#x20;                       cleanLength++
&#x20;                   ] =
&#x20;                       static\_cast\<char>(
&#x20;                           c
&#x20;                           );
&#x20;               }
&#x20;           }

&#x20;           cleanSeed[
&#x20;               cleanLength
&#x20;           ] =
&#x20;               '\0';

&#x20;               if (
&#x20;                   seedEdit
&#x20;                   )
&#x20;               {
&#x20;                   SetWindowTextA(
&#x20;                       seedEdit,
&#x20;                       cleanSeed
&#x20;                   );
&#x20;               }

&#x20;               std::mt19937 seedGenerator(
&#x20;                   hashSeedText(
&#x20;                       cleanSeed
&#x20;                   )
&#x20;               );

&#x20;               std::uniform\_int\_distribution\<int>
&#x20;                   randomAttraction(
&#x20;                       -100,
&#x20;                       100
&#x20;                   );

&#x20;               std::uniform\_int\_distribution\<int>
&#x20;                   randomRange(
&#x20;                       0,
&#x20;                       300
&#x20;                   );

&#x20;               for (
&#x20;                   int from = 0;
&#x20;                   from < COLOR\_COUNT;
&#x20;                   ++from
&#x20;                   )
&#x20;               {
&#x20;                   for (
&#x20;                       int to = 0;
&#x20;                       to < COLOR\_COUNT;
&#x20;                       ++to
&#x20;                       )
&#x20;                   {
&#x20;                       attractionSettings[
&#x20;                           from
&#x20;                       ][
&#x20;                           to
&#x20;                       ] =
&#x20;                               static\_cast\<double>(
&#x20;                                   randomAttraction(
&#x20;                                       seedGenerator
&#x20;                                   )
&#x20;                                   );

&#x20;                           attractionRangeSettings[
&#x20;                               from
&#x20;                           ][
&#x20;                               to
&#x20;                           ] =
&#x20;                                   static\_cast\<double>(
&#x20;                                       randomRange(
&#x20;                                           seedGenerator
&#x20;                                       )
&#x20;                                       );
&#x20;                   }
&#x20;               }

&#x20;               updateGPUSettings();

&#x20;               loadColorPairSettings();

&#x20;               InvalidateRect(
&#x20;                   mainWindow,
&#x20;                   nullptr,
&#x20;                   FALSE
&#x20;               );
}

&#x20;       std::uniform\_int\_distribution\<int>

&#x20;               currentAttraction
&#x20;           );

&#x20;           setButtonText(
&#x20;               BUTTON\_ATTRACTION,
&#x20;               text
&#x20;           );

&#x20;           return 0;
&#x20;       }

&#x20;       if (
&#x20;           sliderID ==
&#x20;           SLIDER\_ATTRACTIONRANGE
&#x20;           )
&#x20;       {
&#x20;           currentAttractionRange =
&#x20;               static\_cast\<double>(
&#x20;                   SendMessage(
&#x20;                       slider,

&#x20;                       TBM\_GETPOS,

&#x20;                       0,
&#x20;                       0
&#x20;                   )
&#x20;                   );

&#x20;           saveColorPairSettings();

&#x20;           char text[64];

&#x20;           sprintf\_s(
&#x20;               text,

&#x20;               sizeof(
&#x20;                   text
&#x20;                   ),

&#x20;               "Range: %.0f",

&#x20;               currentAttractionRange
&#x20;           );

&#x20;           setButtonText(
&#x20;               BUTTON\_ATTRACTIONRANGE,
&#x20;               text
&#x20;           );

&#x20;           return 0;
&#x20;       }

&#x20;       return 0;
&#x20;   }

&#x20;   // ========================================================
&#x20;   // TIMER
&#x20;   // ========================================================

&#x20;   case WM\_TIMER:
&#x20;   {
&#x20;       if (
&#x20;           wParam ==
&#x20;           SIMULATION\_TIMER
&#x20;           )
&#x20;       {
&#x20;           updateSimulation();
&#x20;       }

&#x20;       return 0;
&#x20;   }

&#x20;   // ========================================================
&#x20;   // BACKGROUND
&#x20;   // ========================================================

&#x20;   case WM\_ERASEBKGND:
&#x20;   {
&#x20;       return 1;
&#x20;   }

&#x20;   // ========================================================
&#x20;   // RIGHT MOUSE
&#x20;   // ========================================================

&#x20;   case WM\_RBUTTONDOWN:
&#x20;   {
&#x20;       rightDragging =
&#x20;           true;

&#x20;       lastMouseX =
&#x20;           GET\_X\_LPARAM(
&#x20;               lParam
&#x20;           );

&#x20;       lastMouseY =
&#x20;           GET\_Y\_LPARAM(
&#x20;               lParam
&#x20;           );

&#x20;       SetCapture(
&#x20;           hwnd
&#x20;       );

&#x20;       return 0;
&#x20;   }

&#x20;   case WM\_MOUSEMOVE:
&#x20;   {
&#x20;       if (
&#x20;           rightDragging
&#x20;           )
&#x20;       {
&#x20;           int mouseX =
&#x20;               GET\_X\_LPARAM(
&#x20;                   lParam
&#x20;               );

&#x20;           int mouseY =
&#x20;               GET\_Y\_LPARAM(
&#x20;                   lParam
&#x20;               );

&#x20;           int dx =
&#x20;               mouseX -
&#x20;               lastMouseX;

&#x20;           int dy =
&#x20;               mouseY -
&#x20;               lastMouseY;

&#x20;           cameraX -=
&#x20;               dx /
&#x20;               zoom;

&#x20;           cameraY -=
&#x20;               dy /
&#x20;               zoom;

&#x20;           lastMouseX =
&#x20;               mouseX;

&#x20;           lastMouseY =
&#x20;               mouseY;

&#x20;           InvalidateRect(
&#x20;               hwnd,
&#x20;               nullptr,
&#x20;               FALSE
&#x20;           );
&#x20;       }

&#x20;       return 0;
&#x20;   }

&#x20;   case WM\_RBUTTONUP:
&#x20;   {
&#x20;       rightDragging =
&#x20;           false;

&#x20;       ReleaseCapture();

&#x20;       return 0;
&#x20;   }

&#x20;   // ========================================================
&#x20;   // ZOOM
&#x20;   // ========================================================

&#x20;   case WM\_MOUSEWHEEL:
&#x20;   {
&#x20;       double oldZoom =
&#x20;           zoom;

&#x20;       POINT point =
&#x20;       {
&#x20;           GET\_X\_LPARAM(
&#x20;               lParam
&#x20;           ),

&#x20;           GET\_Y\_LPARAM(
&#x20;               lParam
&#x20;           )
&#x20;       };

&#x20;       ScreenToClient(
&#x20;           hwnd,
&#x20;           &point
&#x20;       );

&#x20;       RECT rect;

&#x20;       GetClientRect(
&#x20;           hwnd,
&#x20;           &rect
&#x20;       );

&#x20;       int width =
&#x20;           rect.right -
&#x20;           rect.left;

&#x20;       int height =
&#x20;           rect.bottom -
&#x20;           rect.top;

&#x20;       double mouseWorldX =
&#x20;           cameraX +
&#x20;           (
&#x20;               point.x -
&#x20;               width /
&#x20;               2.0
&#x20;               ) /
&#x20;           oldZoom;

&#x20;       double mouseWorldY =
&#x20;           cameraY +
&#x20;           (
&#x20;               point.y -
&#x20;               height /
&#x20;               2.0
&#x20;               ) /
&#x20;           oldZoom;

&#x20;       int wheelDelta =
&#x20;           GET\_WHEEL\_DELTA\_WPARAM(
&#x20;               wParam
&#x20;           );

&#x20;       if (
&#x20;           wheelDelta >
&#x20;           0
&#x20;           )
&#x20;       {
&#x20;           zoom \*=
&#x20;               1.15;
&#x20;       }
&#x20;       else if (
&#x20;           wheelDelta <
&#x20;           0
&#x20;           )
&#x20;       {
&#x20;           zoom /=
&#x20;               1.15;
&#x20;       }

&#x20;       zoom =
&#x20;           std::clamp(
&#x20;               zoom,

&#x20;               0.05,

&#x20;               20.0
&#x20;           );

&#x20;       cameraX =
&#x20;           mouseWorldX -
&#x20;           (
&#x20;               point.x -
&#x20;               width /
&#x20;               2.0
&#x20;               ) /
&#x20;           zoom;

&#x20;       cameraY =
&#x20;           mouseWorldY -
&#x20;           (
&#x20;               point.y -
&#x20;               height /
&#x20;               2.0
&#x20;               ) /
&#x20;           zoom;

&#x20;       InvalidateRect(
&#x20;           hwnd,

&#x20;           nullptr,

&#x20;           FALSE
&#x20;       );

&#x20;       return 0;
&#x20;   }

&#x20;   // ========================================================
&#x20;   // PAINT
&#x20;   // ========================================================

&#x20;   case WM\_PAINT:
&#x20;   {
&#x20;       PAINTSTRUCT ps;

&#x20;       BeginPaint(
&#x20;           hwnd,
&#x20;           &ps
&#x20;       );

&#x20;       renderFrame();

&#x20;       EndPaint(
&#x20;           hwnd,
&#x20;           &ps
&#x20;       );

&#x20;       return 0;
&#x20;   }

&#x20;   // ========================================================
&#x20;   // RESIZE
&#x20;   // ========================================================

&#x20;   case WM\_SIZE:
&#x20;   {
&#x20;       if (
&#x20;           gpuInitialized
&#x20;           )
&#x20;       {
&#x20;           int width =
&#x20;               LOWORD(
&#x20;                   lParam
&#x20;               );

&#x20;           int height =
&#x20;               HIWORD(
&#x20;                   lParam
&#x20;               );

&#x20;           if (
&#x20;               width >
&#x20;               0 &&
&#x20;               height >
&#x20;               0
&#x20;               )
&#x20;           {
&#x20;               resizeD3D(
&#x20;                   width,
&#x20;                   height
&#x20;               );
&#x20;           }
&#x20;       }

&#x20;       InvalidateRect(
&#x20;           hwnd,
&#x20;           nullptr,
&#x20;           FALSE
&#x20;       );

&#x20;       return 0;
&#x20;   }

&#x20;   // ========================================================
&#x20;   // DESTROY
&#x20;   // ========================================================

&#x20;   case WM\_DESTROY:
&#x20;   {
&#x20;       KillTimer(
&#x20;           hwnd,
&#x20;           SIMULATION\_TIMER
&#x20;       );

&#x20;       releaseD3D();

&#x20;       PostQuitMessage(
&#x20;           0
&#x20;       );

&#x20;       return 0;
&#x20;   }
&#x20;   }

&#x20;   return DefWindowProc(
&#x20;       hwnd,

&#x20;       message,

&#x20;       wParam,

&#x20;       lParam
&#x20;   );
}
